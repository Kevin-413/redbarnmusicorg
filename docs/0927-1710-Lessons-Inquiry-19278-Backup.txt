document.addEventListener('input', function(e) {
  if (!e.target.matches('input[name="phone-1"]')) return;
  var v = e.target.value.replace(/D/g, '');
  if (v.length === 11 && v.charAt(0) === '1') v = v.substring(1);
  v = v.substring(0, 10);
  var area   = v.substring(0, 3);
  var prefix = v.substring(3, 6);
  var line   = v.substring(6, 10);
  if (v.length >= 7)      e.target.value = '(' + area + ') ' + prefix + '-' + line;
  else if (v.length >= 4) e.target.value = '(' + area + ') ' + prefix;
  else if (v.length >= 1) e.target.value = '(' + area;
});
